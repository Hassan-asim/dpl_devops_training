import base64, datetime, json
from cryptography.hazmat.primitives import serialization, hashes
from cryptography.hazmat.primitives.asymmetric import padding

# Parameters
key_pair_id = "KBVCZWF5LVCT2"
private_key_file = "/tmp/private_key.pem"
url = "https://d2g8ipkslr9kml.cloudfront.net/*"
expire = int((datetime.datetime.utcnow() + datetime.timedelta(minutes=10)).timestamp())

# Load private key
with open(private_key_file, "rb") as f:
    private_key = serialization.load_pem_private_key(f.read(), password=None)

# Create policy
policy = {
    "Statement": [
        {
            "Resource": url,
            "Condition": {
                "DateLessThan": {"AWS:EpochTime": expire}
            }
        }
    ]
}
policy_json = json.dumps(policy).replace(" ", "")

# Sign policy
signature = base64.b64encode(
    private_key.sign(
        policy_json.encode(),
        padding.PKCS1v15(),
        hashes.SHA1()
    )
).decode()

# Output cookies
print("CloudFront-Policy=" + base64.b64encode(policy_json.encode()).decode() + ";")
print("CloudFront-Signature=" + signature + ";")
print("CloudFront-Key-Pair-Id=" + key_pair_id + ";")
EOF
