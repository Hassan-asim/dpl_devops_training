
UPDATE tbl_report_queries

FROM (
    VALUES (
        $$SELECT 
    r.id AS regionalOfficeId,
    r.region AS regionalOffice,
    
    -- 1. COMPLAINT_NOT_FORWARDED
    (
        SELECT ARRAY_AGG(DISTINCT c.id ORDER BY c.id) FILTER (WHERE c.id IS NOT NULL)
        FROM tbl_complain c
        JOIN tbl_user u ON c.region_id = u.region_id
        WHERE c.region_id = r.id
            AND c.id IS NOT NULL
            AND c.examination_date IS NULL 
            AND (u.assign_new_or_transferred_cases IS TRUE OR u.is_registrar IS TRUE)
            AND c.is_active IS TRUE
            AND u.is_active IS TRUE
            AND c.created_date IS NOT NULL
            AND (c.complain_status IS NULL OR c.complain_status NOT IN ('Disposed Off (Implementation completed)', 'Disposed Off (Non-Admissible)', 'Disposed Off (Closed)', 'Disposed Off (Rejected)', 'Disposed Off (Relief Provided)', 'Closed (Rejected by Ombudsman)'))
            AND c.created_date < (
                SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'
                FROM generate_series(
                    (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '10 day',
                    (NOW() AT TIME ZONE 'UTC')::date,
                    '1 day'
                ) AS d
                WHERE EXTRACT(ISODOW FROM d) < 6
                ORDER BY d DESC
                OFFSET 3
                LIMIT 1
            )
    ) AS examinationDelayComplaintIds,
    
    -- 2. IO_REMINDER_REPORT_NOT_CALLED
    (
        SELECT ARRAY_AGG(DISTINCT c.id ORDER BY c.id) FILTER (WHERE c.id IS NOT NULL)
        FROM tbl_complain c
		JOIN tbl_complain_type ct ON ct.id = c.complain_type_id
        JOIN tbl_user u ON c.assign_officer_id = u.id
        WHERE c.region_id = r.id
            AND c.id IS NOT NULL
            AND c.is_active IS TRUE
            AND u.is_active IS TRUE
            AND c.examination_date IS NOT NULL
			AND ct.name <> 'Complaint u/s 33'

                SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'
                FROM generate_series(
                    (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '12 day',
                    (NOW() AT TIME ZONE 'UTC')::date,
                    '1 day'
                ) AS d
                WHERE EXTRACT(ISODOW FROM d) < 6
                ORDER BY d DESC
                OFFSET 7
                LIMIT 1
            )
            AND NOT EXISTS (
                SELECT 1
                FROM tbl_complain_investigation_action ia
                WHERE ia.complain_id = c.id
                  AND ia.is_active IS TRUE
                  AND ia.created_at > c.examination_date
            )
    ) AS reportNotCalledComplaintIds,
    
    -- 3. AGENCY_REPORT_DUE

                    AND ia.is_active IS TRUE

                        SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'

                            (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '12 day',
                            (NOW() AT TIME ZONE 'UTC')::date,
                            '1 day'
                        ) AS d
                        WHERE EXTRACT(ISODOW FROM d) < 6
                        ORDER BY d DESC
                        OFFSET 7
                        LIMIT 1
                    )
            )
    ) AS agencyReportDueComplaintIds,
    
    -- 4. AGENCY_FAILED_SUBMIT_REPORT

        FROM tbl_complain c
		JOIN tbl_complain_type ct ON ct.id = c.complain_type_id
        JOIN tbl_user u ON c.assign_officer_id = u.id
        WHERE c.region_id = r.id

			AND ct.name <> 'Complaint u/s 33'

            )

                WHERE ia.complain_id = c.id
                    AND ia.is_active IS TRUE 
                    AND ia.issue_to = 'agency'
                    AND ia.created_at IS NOT NULL
                    AND ia.created_at < (
                        SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'
                        FROM generate_series(
                            (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '15 day',
                            (NOW() AT TIME ZONE 'UTC')::date,
                            '1 day'
                        ) AS d
                        WHERE EXTRACT(ISODOW FROM d) < 6
                        ORDER BY d DESC
                        OFFSET 10
                        LIMIT 1
                    )
            )
    ) AS noticeNotIssuedComplaintIds,
    
    -- 5. REJOINDER_PENDING_REPORT_NOT_SENT
    (
        SELECT ARRAY_AGG(DISTINCT c.id ORDER BY c.id) FILTER (WHERE c.id IS NOT NULL)
        FROM tbl_complain c
		JOIN tbl_complain_type ct ON ct.id = c.complain_type_id
        JOIN tbl_user u ON c.assign_officer_id = u.id
        WHERE c.region_id = r.id
            AND c.id IS NOT NULL
            AND c.is_active IS TRUE
            AND u.is_active IS TRUE
			AND ct.name <> 'Complaint u/s 33'
            AND (c.complain_status IS NULL OR c.complain_status NOT IN ('Disposed Off (Implementation completed)', 'Disposed Off (Non-Admissible)', 'Disposed Off (Closed)', 'Disposed Off (Rejected)', 'Disposed Off (Relief Provided)', 'Closed (Rejected by Ombudsman)'))

                    AND ia.created_at IS NOT NULL

                        SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'

                            (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '12 day',
                            (NOW() AT TIME ZONE 'UTC')::date,
                            '1 day'
                        ) AS d
                        WHERE EXTRACT(ISODOW FROM d) < 6
                        ORDER BY d DESC
                        OFFSET 7
                        LIMIT 1
                    )
            )
    ) AS rejoinderNotIssuedComplaintIds,
    
    -- 6. HEARING_NOTICE_REQUIRED
    (
        SELECT ARRAY_AGG(DISTINCT c.id ORDER BY c.id) FILTER (WHERE c.id IS NOT NULL)
        FROM tbl_complain c
		JOIN tbl_complain_type ct ON ct.id = c.complain_type_id
        JOIN tbl_user u ON c.assign_officer_id = u.id
        WHERE c.region_id = r.id
            AND c.id IS NOT NULL
            AND c.is_active IS TRUE
            AND u.is_active IS TRUE
			AND ct.name <> 'Complaint u/s 33'
            AND (c.complain_status IS NULL OR c.complain_status NOT IN ('Disposed Off (Implementation completed)', 'Disposed Off (Non-Admissible)', 'Disposed Off (Closed)', 'Disposed Off (Rejected)', 'Disposed Off (Relief Provided)', 'Closed (Rejected by Ombudsman)'))

                        SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'
                        FROM generate_series(
                            (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '10 day',
                            (NOW() AT TIME ZONE 'UTC')::date,
                            '1 day'
                        ) AS d
                        WHERE EXTRACT(ISODOW FROM d) < 6
                        ORDER BY d DESC
                        OFFSET 3
                        LIMIT 1
                    )
            )
            AND NOT EXISTS (
                SELECT 1 FROM tbl_complain_investigation_action ia
                WHERE ia.complain_id = c.id
                    AND ia.is_active IS TRUE 
                    AND ia.action_id = 1 
                    AND ia.action_type_id = 3
            )
    ) AS hearingNoticeRequiredComplaintIds,
    
    -- 7. OFFICER_DID_NOT_APPEAR
    (
        SELECT ARRAY_AGG(DISTINCT c.id ORDER BY c.id) FILTER (WHERE c.id IS NOT NULL)
        FROM tbl_complain c
		JOIN tbl_complain_type ct ON ct.id = c.complain_type_id
        JOIN tbl_user u ON c.region_id = u.region_id AND u.is_ombudsman IS TRUE
        WHERE c.region_id = r.id
            AND c.id IS NOT NULL
            AND c.is_active IS TRUE
            AND u.is_active IS TRUE
			AND ct.name <> 'Complaint u/s 33'
            AND (c.complain_status IS NULL OR c.complain_status NOT IN ('Disposed Off (Implementation completed)', 'Disposed Off (Non-Admissible)', 'Disposed Off (Closed)', 'Disposed Off (Rejected)', 'Disposed Off (Relief Provided)', 'Closed (Rejected by Ombudsman)'))
            AND EXISTS (
                SELECT 1 FROM tbl_complain_investigation_action ia
                WHERE ia.complain_id = c.id
                    AND ia.issue_to = 'agency'
                    AND ia.is_active IS TRUE
            )
            AND (
                SELECT MAX(ia.created_at)
                FROM tbl_complain_investigation_action ia
                WHERE ia.complain_id = c.id
                    AND ia.issue_to = 'agency'
            ) IS NOT NULL
            AND (
                SELECT MAX(ia.created_at)
                FROM tbl_complain_investigation_action ia
                WHERE ia.complain_id = c.id
                    AND ia.issue_to = 'agency'
            ) <= (
                SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'
                FROM generate_series(
                    (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '22 day',
                    (NOW() AT TIME ZONE 'UTC')::date,
                    '1 day'
                ) AS d
                WHERE EXTRACT(ISODOW FROM d) < 6
                ORDER BY d DESC
                OFFSET 15
                LIMIT 1
            )
            AND NOT EXISTS (
                SELECT 1 FROM tbl_complain_investigation_action ia
                WHERE ia.complain_id = c.id
                    AND ia.issue_to = 'agency'
                    AND ia.action_id = 6
                    AND ia.action_type_id = 22
                    AND ia.is_active IS TRUE
            )
    ) AS escalationOfficerComplaintIds,
    
    -- 8. CRITICAL_ALERT_NOT_RESOLVED
    (
        SELECT ARRAY_AGG(DISTINCT c.id ORDER BY c.id) FILTER (WHERE c.id IS NOT NULL)
        FROM tbl_complain c
        JOIN tbl_user u ON c.assign_officer_id = u.id
        WHERE c.region_id = r.id
            AND c.id IS NOT NULL
            AND c.is_active IS TRUE
            AND u.is_active IS TRUE
            AND c.created_date IS NOT NULL
            AND (c.complain_status IS NULL OR c.complain_status NOT IN ('Disposed Off (Implementation completed)', 'Disposed Off (Non-Admissible)', 'Disposed Off (Closed)', 'Disposed Off (Rejected)', 'Disposed Off (Relief Provided)', 'Closed (Rejected by Ombudsman)'))
            AND c.created_date < (
                SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'
                FROM generate_series(
                    (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '90 day',
                    (NOW() AT TIME ZONE 'UTC')::date,
                    '1 day'
                ) AS d
                WHERE EXTRACT(ISODOW FROM d) < 6
                ORDER BY d DESC
                OFFSET 60
                LIMIT 1
            )
    ) AS criticalAlertComplaintIds,
    
    -- 9. DRAFT_DECISION_DELAY
    (
        SELECT ARRAY_AGG(DISTINCT c.id ORDER BY c.id) FILTER (WHERE c.id IS NOT NULL)
        FROM tbl_complain c
		JOIN tbl_complain_type ct ON ct.id = c.complain_type_id
        JOIN tbl_user u ON c.assign_officer_id = u.id
        WHERE c.region_id = r.id
            AND c.id IS NOT NULL
            AND c.is_active IS TRUE
            AND u.is_active IS TRUE
			AND ct.name <> 'Complaint u/s 33'
            AND (c.complain_status IS NULL OR c.complain_status NOT IN ('Disposed Off (Implementation completed)', 'Disposed Off (Non-Admissible)', 'Disposed Off (Closed)', 'Disposed Off (Rejected)', 'Disposed Off (Relief Provided)', 'Closed (Rejected by Ombudsman)'))
            AND EXISTS (
                SELECT 1 FROM public.tbl_complain_decision_draft cd
                WHERE cd.status = 'submitted' 
                    AND cd.is_active IS TRUE 
                    AND cd.complain_id = c.id 
                    AND cd.created_at IS NOT NULL
                    AND cd.created_at < (
                        SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'
                        FROM generate_series(
                            (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '12 day',
                            (NOW() AT TIME ZONE 'UTC')::date,
                            '1 day'
                        ) AS d
                        WHERE EXTRACT(ISODOW FROM d) < 6
                        ORDER BY d DESC
                        OFFSET 7
                        LIMIT 1
                    )
            )
            AND NOT EXISTS (
                SELECT 1 FROM public.tbl_complain_decision d
                WHERE d.is_active IS TRUE 
                    AND d.complain_id = c.id
            )
    ) AS draftDecisionDelayComplaintIds,
    
    -- 10. IMPLEMENTATION_DELAY
    (
        SELECT ARRAY_AGG(DISTINCT c.id ORDER BY c.id) FILTER (WHERE c.id IS NOT NULL)
        FROM tbl_complain c
		JOIN tbl_complain_type ct ON ct.id = c.complain_type_id
        JOIN tbl_user u ON c.assign_officer_id = u.id
        WHERE c.region_id = r.id
            AND c.id IS NOT NULL
            AND c.is_active IS TRUE
            AND u.is_active IS TRUE
            AND c.created_date IS NOT NULL
			AND ct.name <> 'Complaint u/s 33'
            AND (c.complain_status IS NULL OR c.complain_status NOT IN ('Disposed Off (Implementation completed)', 'Disposed Off (Non-Admissible)', 'Disposed Off (Closed)', 'Disposed Off (Rejected)', 'Disposed Off (Relief Provided)', 'Closed (Rejected by Ombudsman)'))
            AND c.created_date < (
                SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'
                FROM generate_series(
                    (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '260 day',
                    (NOW() AT TIME ZONE 'UTC')::date,
                    '1 day'
                ) AS d
                WHERE EXTRACT(ISODOW FROM d) < 6
                ORDER BY d DESC
                OFFSET 180
                LIMIT 1
            )
    ) AS implementationDelayComplaintIds,
    
    -- 11. SUMMON_REQUIRED
    (
        SELECT ARRAY_AGG(DISTINCT c.id ORDER BY c.id) FILTER (WHERE c.id IS NOT NULL)
        FROM tbl_complain c
		JOIN tbl_complain_type ct ON ct.id = c.complain_type_id
        JOIN tbl_user u ON c.assign_officer_id = u.id
        WHERE c.region_id = r.id

            AND u.is_active IS TRUE
			AND ct.name <> 'Complaint u/s 33'

            AND EXISTS (
                SELECT 1 FROM tbl_complain_investigation_action ia

                    AND ia.issue_to = 'agency'

            AND (
                SELECT MAX(ia.created_at)
                FROM tbl_complain_investigation_action ia

                    AND ia.issue_to = 'agency'

                    (NOW() AT TIME ZONE 'UTC')::date,
                    '1 day'
                ) AS d

                ORDER BY d DESC

                WHERE ia.complain_id = c.id

FROM tbl_region r
WHERE r.is_active = TRUE
ORDER BY r.region;$$
    )
) AS q(updated_query)
WHERE tbl_report_queries.entity_type = 'ESCALATION'
  AND tbl_report_queries.query_name   = 'REGIONAL_ESCALATION_REPORT';
