UPDATE tbl_escalations
SET query_sql = $$
    SELECT 
        c.diary_number          AS ComplainNumber,
        u.email                 AS Receiver,
        u.id                    AS UserId
    FROM tbl_complain c
    JOIN tbl_user u 
        ON c.assign_officer_id = u.id
    WHERE 
        c.is_active IS TRUE
        AND u.is_active IS TRUE
        AND c.examination_date < (
            SELECT (d::date + TIME '00:00') AT TIME ZONE 'UTC'
            FROM generate_series(
                     (NOW() AT TIME ZONE 'UTC')::date - INTERVAL '21 day',
                     (NOW() AT TIME ZONE 'UTC')::date,
                     '1 day'
                 ) AS d
            WHERE EXTRACT(ISODOW FROM d) < 6          -- Mon–Fri (ISODOW 1–5)
            ORDER BY d DESC
            OFFSET 7                                  -- skip the most recent 7 working days
            LIMIT 1
        )
        AND NOT EXISTS (
            SELECT 1
            FROM tbl_complain_investigation_action ia
            WHERE ia.complain_id = c.id
              AND ia.is_active IS TRUE
              AND ia.created_at > c.examination_date
        )
        AND NOT EXISTS (
            SELECT 1 
            FROM tbl_escalation_logs el
            WHERE el.complaint_number = c.diary_number
              AND el.escalation_code   = 'IO_REMINDER_REPORT_NOT_CALLED'
              AND el.recipient_user_id::text = u.id::text
              AND DATE(el.execution_date) = CURRENT_DATE
              AND el.escalation_status = 'sent'
        )
        AND NOT EXISTS (
            SELECT 1 
            FROM tbl_complain_type ct
            WHERE ct.id = c.type_of_complaint_id
              AND ct.name = 'Complaint u/s 33'
        )
		 AND NOT EXISTS (
        SELECT 1
        FROM tbl_complain_disposal cd
        JOIN tbl_disposal_reason dr 
            ON cd.disposal_reason_id = dr.id
        WHERE cd.complain_id = c.id
          AND dr.name = 'With Directions'
    )
    LIMIT 10;
$$
WHERE escalation_code = 'IO_REMINDER_REPORT_NOT_CALLED'
;
